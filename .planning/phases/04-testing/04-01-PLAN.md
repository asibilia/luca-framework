**Files**: Create `__tests__/utils/mock-execa.ts`
**Details**:

```typescript
/**
 * Shared execa mock factory for tests that need to mock shell commands.
 *
 * Usage in tests:
 *   import { createExecaMock, installExecaMock } from '../../utils/mock-execa';
 *   installExecaMock(createExecaMock({ stdout: '{"number": 1}' }));
 *   const { myFunction } = await import('../path/to/module');
 */
import { mock } from 'bun:test';

export interface ExecaCallConfig {
  stdout?: string;
  stderr?: string;
  exitCode?: number;
  error?: Error;
}

/**
 * Create a mock execa function with configurable behavior.
 *
 * @param defaultConfig - Default response for all calls
 * @param callConfigs - Per-call overrides (indexed by call order)
 */
export function createExecaMock(
  defaultConfig: ExecaCallConfig = { stdout: '', stderr: '', exitCode: 0 },
  callConfigs?: ExecaCallConfig[]
) {
  let callIndex = 0;
  const calls: Array<{ command: string; args: string[] }> = [];

  const execaMock = async (command: string, args: string[] = []) => {
    calls.push({ command, args });
    const config = callConfigs?.[callIndex] ?? defaultConfig;
    callIndex++;

    if (config.error) {
      throw config.error;
    }

    return {
      stdout: config.stdout ?? '',
      stderr: config.stderr ?? '',
      exitCode: config.exitCode ?? 0,
    };
  };

  return { execa: execaMock, getCalls: () => calls };
}

/**
 * Install execa mock via bun:test mock.module().
 * MUST be called before importing any module that uses execa.
 */
export function installExecaMock(mockInstance: ReturnType<typeof createExecaMock>) {
  mock.module('execa', () => ({
    execa: mockInstance.execa,
  }));
}
```

### Task 5: Create shared mock factory -- `__tests__/utils/mock-fetch.ts`
**Goal**: Provide a reusable fetch mock factory for Jira adapter tests.
**Files**: Create `__tests__/utils/mock-fetch.ts`
**Details**:

```typescript
/**
 * Shared fetch mock factory for tests that need to mock HTTP requests.
 *
 * Usage in tests:
 *   import { createFetchMock, installFetchMock } from '../../utils/mock-fetch';
 *   installFetchMock(createFetchMock({ status: 200, body: { key: 'PROJ-123' } }));
 */

export interface FetchMockConfig {
  status?: number;
  body?: unknown;
  headers?: Record<string, string>;
  error?: Error;
}

/**
 * Create a mock fetch function with configurable behavior.
 *
 * @param defaultConfig - Default response for all calls
 * @param urlConfigs - URL-pattern-specific overrides (key is substring to match in URL)
 */
export function createFetchMock(
  defaultConfig: FetchMockConfig = { status: 200, body: {} },
  urlConfigs?: Record<string, FetchMockConfig>
) {
  const calls: Array<{ url: string; options?: RequestInit }> = [];

  const fetchMock = async (url: string | URL | Request, options?: RequestInit) => {
    const urlStr = typeof url === 'string' ? url : url instanceof URL ? url.toString() : url.url;
    calls.push({ url: urlStr, options });

    // Check for URL-specific config
    let config = defaultConfig;
    if (urlConfigs) {
      for (const [pattern, patternConfig] of Object.entries(urlConfigs)) {
        if (urlStr.includes(pattern)) {
          config = patternConfig;
          break;
        }
      }
    }

    if (config.error) {
      throw config.error;
    }

    return new Response(JSON.stringify(config.body), {
      status: config.status ?? 200,
      headers: {
        'Content-Type': 'application/json',
        ...(config.headers ?? {}),
      },
    });
  };

  return { fetch: fetchMock, getCalls: () => calls };
}

/**
 * Install fetch mock by replacing globalThis.fetch.
 * Returns a restore function to call in afterEach.
 */
export function installFetchMock(mockInstance: ReturnType<typeof createFetchMock>): () => void {
  const originalFetch = globalThis.fetch;
  globalThis.fetch = mockInstance.fetch as typeof fetch;
  return () => {
    globalThis.fetch = originalFetch;
  };
}
```

/**
 * Shared @clack/prompts mock factory for tests that need to mock interactive prompts.
 *
 * Usage in tests:
 *   import { installClackMock, createWizardResponses } from '../../utils/mock-clack';
 *   installClackMock(createWizardResponses({ cancel: false }));
 */
import { mock } from 'bun:test';

export interface ClackMockConfig {
  /** Responses for p.group() calls -- returns the object as-is */
  groupResponse?: Record<string, unknown> | null;
  /** Responses for p.select() calls -- returns values in order */
  selectResponses?: (string | symbol)[];
  /** Response for p.confirm() */
  confirmResponse?: boolean | symbol;
  /** Response for p.text() */
  textResponses?: (string | symbol)[];
}

const CANCEL_SYMBOL = Symbol.for('cancel');

/**
 * Create a complete wizard response set that simulates a user completing all prompts.
 */
export function createWizardResponses(options: {
  cancel?: false;
  frameworkName?: string;
  commandPrefix?: string;
  stack?: string;
  workTracker?: string;
}): ClackMockConfig {
  return {
    groupResponse: {
      frameworkName: options.frameworkName ?? 'Luca',
      commandPrefix: options.commandPrefix ?? 'lu',
      ticketPattern: '[A-Z]+-\\d+',
      placeholderTicket: 'PROJ-0000',
    },
    selectResponses: [options.stack ?? 'custom', options.workTracker ?? 'none'],
    confirmResponse: true,
  };
}

/**
 * Create a wizard response set that cancels at a specific step.
 */
export function createCancelledWizardResponses(cancelAt: 'group' | 'stack' | 'tracker' | 'confirm'): ClackMockConfig {
  if (cancelAt === 'group') return { groupResponse: null };
  if (cancelAt === 'stack') return { groupResponse: { frameworkName: 'Luca', commandPrefix: 'lu', ticketPattern: '[A-Z]+-\\d+', placeholderTicket: 'PROJ-0000' }, selectResponses: [CANCEL_SYMBOL] };
  if (cancelAt === 'tracker') return { groupResponse: { frameworkName: 'Luca', commandPrefix: 'lu', ticketPattern: '[A-Z]+-\\d+', placeholderTicket: 'PROJ-0000' }, selectResponses: ['custom', CANCEL_SYMBOL] };
  // cancelAt === 'confirm'
  return { groupResponse: { frameworkName: 'Luca', commandPrefix: 'lu', ticketPattern: '[A-Z]+-\\d+', placeholderTicket: 'PROJ-0000' }, selectResponses: ['custom', 'none'], confirmResponse: CANCEL_SYMBOL as unknown as boolean };
}

/**
 * Install @clack/prompts mock via bun:test mock.module().
 * MUST be called before importing any module that uses @clack/prompts.
 */
export function installClackMock(config: ClackMockConfig) {
  let selectCallIndex = 0;
  let textCallIndex = 0;

  mock.module('@clack/prompts', () => ({
    intro: () => {},
    outro: () => {},
    cancel: () => {},
    note: () => {},
    log: { info: () => {}, warn: () => {}, error: () => {} },
    spinner: () => ({
      start: () => {},
      stop: () => {},
    }),
    isCancel: (value: unknown) => typeof value === 'symbol',
    group: async () => config.groupResponse,
    select: async () => {
      const response = config.selectResponses?.[selectCallIndex] ?? 'default';
      selectCallIndex++;
      return response;
    },
    confirm: async () => config.confirmResponse ?? true,
    text: async () => {
      const response = config.textResponses?.[textCallIndex] ?? 'default';
      textCallIndex++;
      return response;
    },
  }));
}

### Task 7: Create shared temp directory helper -- `__tests__/utils/temp-dir.ts`
**Goal**: Provide helpers for creating and cleaning up temporary directories for filesystem tests.
**Files**: Create `__tests__/utils/temp-dir.ts`
**Details**:

```typescript
/**
 * Temporary directory helpers for filesystem tests.
 *
 * Usage in tests:
 *   import { createTempDir, cleanupTempDir, setupTempProject } from '../../utils/temp-dir';
 *
 *   let tempDir: string;
 *   beforeEach(async () => { tempDir = await createTempDir(); });
 *   afterEach(async () => { await cleanupTempDir(tempDir); });
 */
import { mkdtemp, rm, writeFile, mkdir } from 'fs/promises';
import { join } from 'path';
import { tmpdir } from 'os';

/**
 * Create a unique temporary directory for a test.
 * @param prefix - Directory prefix (default: 'luca-test-')
 * @returns Absolute path to the created directory
 */
export async function createTempDir(prefix = 'luca-test-'): Promise<string> {
  return mkdtemp(join(tmpdir(), prefix));
}

/**
 * Remove a temporary directory and all contents.
 * @param dirPath - Directory to remove
 */
export async function cleanupTempDir(dirPath: string): Promise<void> {
  try {
    await rm(dirPath, { recursive: true, force: true });
  } catch {
    // Ignore cleanup errors
  }
}

/**
 * Create a temporary project directory with specified file structure.
 *
 * @param files - Map of relative path to file content
 * @returns Absolute path to the created directory
 *
 * @example
 * ```typescript
 * const dir = await setupTempProject({
 *   'package.json': JSON.stringify({ name: 'test-project' }),
 *   '.planning/config.json': JSON.stringify({ branding: {...} }),
 *   '.git/.gitkeep': '',
 * });
 * ```
 */
export async function setupTempProject(files: Record<string, string>): Promise<string> {
  const dir = await createTempDir();

  for (const [relativePath, content] of Object.entries(files)) {
    const fullPath = join(dir, relativePath);
    const parentDir = fullPath.substring(0, fullPath.lastIndexOf('/'));
    await mkdir(parentDir, { recursive: true });
    await writeFile(fullPath, content, 'utf-8');
  }

  return dir;
}
```
