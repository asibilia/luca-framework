# Phase 03 Plan 01: MuninnDB API Layer — 8 New Routes + Enhanced Filtering

## Objective

Build 8 new Next.js API routes that proxy MuninnDB operations, plus enhance the existing `/api/muninn/engrams` route with type/tag/entity/since filtering. This gives the observer frontend complete data access for Phases 04-07 views (Entity Deep Dive, Knowledge Graph, Session Explorer, Contradiction View).

The critical constraint: only 3 of the 8 target endpoints have direct MuninnDB REST equivalents. The other 5 must be composed from available REST primitives (engrams listing with tags filter, engram links lookup).

// entity — compose from engrams (tags filter) + links for each engram
async entity(vault: string, name: string, limit = 20) {
  // Step 1: Fetch engrams tagged with entity name
  const engramsRes = await muninnFetch(
    `/api/engrams?vault=${encodeURIComponent(vault)}&tags=${encodeURIComponent(name)}&limit=${limit}`,
  );
  if (!engramsRes.ok) throw new Error(`MuninnDB entity engrams: ${engramsRes.status}`);
  const engramsData = await engramsRes.json();
  const engrams = engramsData.engrams ?? [];

  // Step 2: Fetch links for the first engram (to get relationships)
  let relationships: unknown[] = [];
  if (engrams.length > 0) {
    try {
      const linksRes = await muninnFetch(`/api/engrams/${engrams[0].id}/links`);
      if (linksRes.ok) {
        const linksData = await linksRes.json();
        relationships = linksData.associations ?? linksData.links ?? [];
      }
    } catch { /* links fetch is best-effort */ }
  }

  return {
    name,
    type: "unknown",
    confidence: 1,
    state: "active",
    mention_count: engrams.length,
    first_seen: engrams.length > 0 ? engrams[engrams.length - 1].created_at : null,
    updated_at: engrams.length > 0 ? engrams[0].created_at : null,
    engrams: engrams.map((e: Record<string, unknown>) => ({
      id: e.id,
      concept: e.concept,
      created_at: e.created_at,
    })),
    relationships,
    co_occurring: [],
  };
},

// entityTimeline — same as entity but sorted chronologically
async entityTimeline(vault: string, entityName: string, limit = 50) {
  const engramsRes = await muninnFetch(
    `/api/engrams?vault=${encodeURIComponent(vault)}&tags=${encodeURIComponent(entityName)}&limit=${limit}`,
  );
  if (!engramsRes.ok) throw new Error(`MuninnDB entityTimeline: ${engramsRes.status}`);
  const data = await engramsRes.json();
  const engrams = data.engrams ?? [];

  // Sort by created_at ascending (oldest first = timeline order)
  const sorted = [...engrams].sort(
    (a: Record<string, unknown>, b: Record<string, unknown>) =>
      new Date(a.created_at as string).getTime() - new Date(b.created_at as string).getTime(),
  );

  return {
    entity: entityName,
    first_seen: sorted.length > 0 ? sorted[0].created_at : null,
    mention_count: sorted.length,
    timeline: sorted.map((e: Record<string, unknown>) => ({
      engram_id: e.id,
      concept: e.concept,
      created_at: e.created_at,
      summary: e.content ?? "",
    })),
    count: sorted.length,
  };
},

// entityClusters — fetch all engrams, compute tag co-occurrence server-side
async entityClusters(vault: string, topN = 20, minCount = 2) {
  const res = await muninnFetch(
    `/api/engrams?vault=${encodeURIComponent(vault)}&limit=1000`,
  );
  if (!res.ok) throw new Error(`MuninnDB entityClusters: ${res.status}`);
  const data = await res.json();
  const engrams = data.engrams ?? [];

  // Compute tag co-occurrence pairs
  const pairCounts = new Map<string, number>();
  for (const engram of engrams) {
    const tags: string[] = engram.tags ?? [];
    for (let i = 0; i < tags.length; i++) {
      for (let j = i + 1; j < tags.length; j++) {
        const pair = [tags[i], tags[j]].sort().join("|||");
        pairCounts.set(pair, (pairCounts.get(pair) ?? 0) + 1);
      }
    }
  }

  // Filter by minCount and sort by count descending
  const clusters = Array.from(pairCounts.entries())
    .filter(([, count]) => count >= minCount)
    .sort((a, b) => b[1] - a[1])
    .slice(0, topN)
    .map(([pair, count]) => {
      const [entity_a, entity_b] = pair.split("|||");
      return { entity_a, entity_b, count };
    });

  return { clusters, count: clusters.length };
},

// exportGraph — fetch all engrams + links, assemble JSON-LD
async exportGraph(vault: string, format = "json-ld", includeEngrams = false) {
  const res = await muninnFetch(
    `/api/engrams?vault=${encodeURIComponent(vault)}&limit=1000`,
  );
  if (!res.ok) throw new Error(`MuninnDB exportGraph: ${res.status}`);
  const data = await res.json();
  const engrams = data.engrams ?? [];

  // Collect unique entities from tags
  const entitySet = new Set<string>();
  for (const engram of engrams) {
    for (const tag of (engram.tags ?? [])) {
      entitySet.add(tag);
    }
  }

  // Build JSON-LD graph
  const graph: Record<string, unknown>[] = Array.from(entitySet).map((name) => ({
    "@id": `entity:${name}`,
    "@type": "Entity",
    name,
  }));

  if (includeEngrams) {
    for (const engram of engrams) {
      graph.push({
        "@id": `engram:${engram.id}`,
        "@type": "Engram",
        concept: engram.concept,
        created_at: engram.created_at,
      });
    }
  }

  const jsonLd = JSON.stringify({
    "@context": { "@vocab": "https://muninndb.com/schema/" },
    "@graph": graph,
  });

  return {
    data: jsonLd,
    node_count: entitySet.size + (includeEngrams ? engrams.length : 0),
    edge_count: 0,
    format,
  };
},

```typescript
import {
  muninnProxyHandler,
  parseQueryParams,
} from "~/lib/muninn-route-helper";
import {
  EngramsQuerySchema,
  EngramsResponseSchema,
} from "~/lib/muninn-schemas";

/**
 * GET /api/muninn/engrams
 *
 * Proxies MuninnDB engram listing with optional filtering.
 * Query params:
 * - vault (default: "default")
 * - limit (default: 100, max: 1000)
 * - offset (default: 0)
 * - tag (optional) — server-side filter via MuninnDB tags param
 * - type (optional) — client-side filter by memory_type
 * - entity (optional) — client-side filter by entity mention in content
 * - since (optional) — client-side filter by created_at >= timestamp
 */
export async function GET(request: Request) {
  const { searchParams } = new URL(request.url);
  const result = parseQueryParams(searchParams, EngramsQuerySchema);
  if (!result.success) return result.response;

  const { vault, limit, offset, tag, type, entity, since } = result.data;

  // If client-side filters are active, fetch more data to ensure adequate results
  const hasClientFilters = type || entity || since;
  const fetchLimit = hasClientFilters ? Math.min(limit * 5, 1000) : limit;

  return muninnProxyHandler(
    async (client) => {
      const data = await client.listEngrams(vault, fetchLimit, offset, tag);
      if (!hasClientFilters) return data;

      // Apply client-side filters
      let filtered = data.engrams ?? [];
      if (type) {
        filtered = filtered.filter(
          (e: Record<string, unknown>) => e.memory_type === type,
        );
      }
      if (entity) {
        const entityLower = entity.toLowerCase();
        filtered = filtered.filter(
          (e: Record<string, unknown>) =>
            String(e.content ?? "")
              .toLowerCase()
              .includes(entityLower) ||
            (Array.isArray(e.tags) &&
              e.tags.some((t: string) =>
                t.toLowerCase().includes(entityLower),
              )),
        );
      }
      if (since) {
        filtered = filtered.filter((e: Record<string, unknown>) => {
          const createdAt =
            typeof e.created_at === "number"
              ? e.created_at
              : new Date(e.created_at as string).getTime() / 1000;
          return createdAt >= since;
        });
      }

      return { engrams: filtered.slice(0, limit), total: filtered.length };
    },
    "Failed to fetch engrams from MuninnDB",
    EngramsResponseSchema,
  );
}
```

import { muninnProxyHandler } from "~/lib/muninn-route-helper";
import {
  TraverseRequestSchema,
  TraverseResponseSchema,
} from "~/lib/muninn-schemas";

/**
 * POST /api/muninn/traverse
 *
 * Proxies MuninnDB graph traversal. Walks the engram association graph
 * from a starting node.
 */
export async function POST(request: Request) {
  let body: unknown;
  try {
    body = await request.json();

    return NextResponse.json({ error: "Invalid JSON body" }, { status: 400 });

  if (!parsed.success) {

    parsed.data;

  return muninnProxyHandler(
    (client) =>
      client.traverse(
        vault,
        start_id,
        max_hops,
        max_nodes,
        follow_entities,
        rel_types,
      ),
    "Failed to traverse MuninnDB graph",
    TraverseResponseSchema,
  );
}
