### 7. Create entity deep dive page and placeholder index

**Type:** auto

**Dynamic route page: `packages/luca-observer/app/entities/[name]/page.tsx`**

Follow the decisions/page.tsx pattern exactly:

```
"use client"

import { useParams } from "next/navigation"
import { useState } from "react"

// Layout imports
import { PageContainer } from "~/components/layout/page-container"
import { ErrorBoundary } from "~/components/shared/error-boundary"
import { LoadingSkeleton } from "~/components/shared/loading-skeleton"

// Entity component imports
import { EntityHeader } from "~/components/entities/entity-header"
import { EntityTabBar } from "~/components/entities/entity-tab-bar"
import type { TabId } from "~/components/entities/entity-tab-bar"
import { EntityTimeline } from "~/components/entities/entity-timeline"
import { EntityRelationships } from "~/components/entities/entity-relationships"
import { EntityEngrams } from "~/components/entities/entity-engrams"
import { EntityCoOccurrences } from "~/components/entities/entity-co-occurrences"

// Hook
import { useEntityDeepDive } from "~/hooks/use-entity-deep-dive"

// Utils
import { relativeTime } from "~/lib/format"
```

**Page implementation:**

1. Extract `name` from `useParams()` -- decode it with `decodeURIComponent(name as string)`
2. Pass decoded entity name to `useEntityDeepDive(entityName)`
3. Local state: `const [activeTab, setActiveTab] = useState<TabId>("timeline")`
4. Build `lastUpdatedText` from `lastUpdated` using `relativeTime`
5. PageContainer with:
   - `title="Entity"` subtitle={entityName}
   - Actions: last updated text + refresh button (same pattern as decisions page)
6. Loading state: show LoadingSkeleton (card + text variants)
7. Error state: show error message in a styled container
8. Loaded state:
   - If entity is null (not found), show EmptyState
   - Otherwise render: EntityHeader, EntityTabBar, then conditionally render the active tab content wrapped in ErrorBoundary
9. Tab content mapping:
   - "timeline" -> `<EntityTimeline timeline={timeline} />`
   - "relationships" -> `<EntityRelationships relationships={entity.relationships} entityName={entityName} />`
   - "engrams" -> `<EntityEngrams engrams={entity.engrams} />`
   - "co-occurrences" -> `<EntityCoOccurrences coOccurrences={coOccurrences} />`
