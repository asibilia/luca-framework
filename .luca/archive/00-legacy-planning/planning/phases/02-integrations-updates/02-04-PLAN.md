```typescript
/**
 * Compare manifest files with current filesystem and new content.
 * 
 * Three-way comparison algorithm:
 * - unchanged: originalHash === currentHash → safe to update
 * - user-modified: originalHash !== currentHash → conflict
 * - new: no originalHash in manifest → add file
 * - deleted: file missing from filesystem → conflict
 * 
 * @param manifest - Existing manifest from project
 * @param newFiles - Map of path → content from new framework version
 * @param cwd - Working directory
 * @returns Array of FileComparison results
 */
export async function compareFiles(
  manifest: LucaManifest,
  newFiles: Map<string, string>,
  cwd: string
): Promise<FileComparison[]> {
  const results: FileComparison[] = [];
  
  for (const [path, newContent] of newFiles) {
    const newHash = createHash('sha256').update(newContent).digest('hex');
    const manifestEntry = manifest.files[path];
    
    if (!manifestEntry) {
      // New file in update
      results.push({
        path,
        status: 'new',
        originalHash: null,
        currentHash: null,
        newHash,
      });
      continue;
    }
    
    const fullPath = join(cwd, path);
    let currentHash: string | null = null;
    
    try {
      currentHash = await hashFile(fullPath);
    } catch {
      // File was deleted by user
      results.push({
        path,
        status: 'deleted',
        originalHash: manifestEntry.originalHash,
        currentHash: null,
        newHash,
      });
      continue;
    }
    
    if (currentHash === manifestEntry.originalHash) {
      // User hasn't modified - safe to update
      results.push({
        path,
        status: 'unchanged',
        originalHash: manifestEntry.originalHash,
        currentHash,
        newHash,
      });
    } else {
      // User has modified - conflict
      results.push({
        path,
        status: 'user-modified',
        originalHash: manifestEntry.originalHash,
        currentHash,
        newHash,
      });
    }
  }
  
  return results;
}
```

import { defineCommand } from 'citty';
import * as p from '@clack/prompts';

export const updateCommand = defineCommand({
  meta: {
    name: 'update',
    description: 'Update Luca framework to the latest version',
  },
  args: {
    force: {
      type: 'boolean',
      description: 'Force update, overwriting user modifications',
      default: false,
    },
    'dry-run': {
      type: 'boolean',
      description: 'Show what would be updated without making changes',
      default: false,
    },
    'accept-theirs': {
      type: 'boolean',
      description: 'Accept all new framework versions for conflicts',
      default: false,
    },
    'accept-mine': {
      type: 'boolean',
      description: 'Keep all user modifications, skip conflicting files',
      default: false,
    },
  },
  async run({ args }) {
    const cwd = process.cwd();
    
    p.intro('Luca Update');
    
    // 1. Check Luca is installed
    const manifest = await readManifest(cwd);
    if (!manifest) {
      logger.error('Luca not installed. Run `npx luca init` first.');
      process.exit(1);
    }
    
    logger.info(`Current version: ${manifest.version}`);
    
    // 2. Get new framework files
    // For now, read from installed package templates
    // Future: fetch from npm registry
    const newFiles = await getNewFrameworkFiles(cwd, manifest);
    
    // 3. Compare files
    const comparisons = await compareFiles(manifest, newFiles, cwd);
    
    const unchanged = comparisons.filter(c => c.status === 'unchanged');
    const modified = comparisons.filter(c => c.status === 'user-modified');
    const newF = comparisons.filter(c => c.status === 'new');
    const deleted = comparisons.filter(c => c.status === 'deleted');
    
    // 4. Show summary
    logger.info(`Files: ${unchanged.length} unchanged, ${newF.length} new, ${modified.length} modified, ${deleted.length} deleted`);
    
    if (args['dry-run']) {
      showDryRunSummary(comparisons);
      p.outro('Dry run complete. No changes made.');
      return;
    }
    
    // 5. Handle conflicts
    if (modified.length > 0 || deleted.length > 0) {
      if (args.force || args['accept-theirs']) {
        logger.warn(`Overwriting ${modified.length + deleted.length} user modifications`);
      } else if (args['accept-mine']) {
        logger.info(`Keeping ${modified.length + deleted.length} user modifications`);
      } else {
        await handleConflicts(modified, deleted, cwd);
      }
    }
    
    // 6. Create backup
    const backupDir = join(cwd, `.luca-backup-${Date.now()}`);
    await createBackup(cwd, manifest, backupDir);
    
    try {
      // 7. Apply updates
      await applyUpdates(comparisons, newFiles, cwd, args);
      
      // 8. Update manifest
      await updateManifestAfterUpdate(manifest, comparisons, newFiles, cwd);
      
      // 9. Clean up backup
      await rm(backupDir, { recursive: true });
      
      p.outro('✅ Update complete!');
    } catch (error) {
      // Restore from backup
      logger.error('Update failed, restoring from backup...');
      await restoreBackup(cwd, backupDir);
      await rm(backupDir, { recursive: true });
      throw error;
    }
  },
});
